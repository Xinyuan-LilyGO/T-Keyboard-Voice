get_filename_component(VOICE_REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(CMAKE_BUILD_EARLY_EXPANSION)
    set(VOICE_EXAMPLE "$ENV{TKEYBOARD_VOICE_EXAMPLE}")
endif()
set(sources "${VOICE_REPO}/src/VoicePower.cpp")
set(requires espressif__arduino-esp32 voice_sensorlib)
if(VOICE_EXAMPLE STREQUAL "Factory")
    list(APPEND sources "${VOICE_REPO}/examples/Factory/Factory.cpp")
else()
    # 把同一份 .ino 作为 C++ 编译；示例已显式包含所需头文件，无须自动原型。
    set(sketch "${VOICE_REPO}/examples/${VOICE_EXAMPLE}/${VOICE_EXAMPLE}.ino")
    if(NOT CMAKE_BUILD_EARLY_EXPANSION)
        set_source_files_properties("${sketch}" PROPERTIES LANGUAGE CXX COMPILE_FLAGS "-x c++")
    endif()
    list(APPEND sources "${sketch}")
endif()
if(VOICE_EXAMPLE MATCHES "^(Display|Factory)$")
    list(APPEND sources "${VOICE_REPO}/src/VoiceDisplay.cpp")
    list(APPEND requires voice_lovyangfx)
endif()
if(VOICE_EXAMPLE MATCHES "^(Keys|Factory)$")
    list(APPEND sources "${VOICE_REPO}/src/VoiceInput.cpp")
    list(APPEND requires voice_buttonsense)
endif()
if(VOICE_EXAMPLE MATCHES "^(WS2812|Factory)$")
    list(APPEND sources "${VOICE_REPO}/src/VoiceRGB.cpp")
endif()
if(VOICE_EXAMPLE MATCHES "^(AudioPlayback|AudioRecord|Factory)$")
    list(APPEND sources "${VOICE_REPO}/src/VoiceAudio.cpp")
endif()
if(VOICE_EXAMPLE MATCHES "^(AudioPlayback|Factory)$")
    list(APPEND sources "${VOICE_REPO}/src/VoiceTestAudio.cpp")
endif()
idf_component_register(SRCS ${sources} INCLUDE_DIRS "${VOICE_REPO}/src" REQUIRES ${requires})
target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra)
